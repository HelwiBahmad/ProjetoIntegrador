  // services/NoticiaService.js
  class NoticiaService {
    static buscarTodas(filtros = {}) {
      // VALIDAÇÕES de negócio
      if (filtros.pagina < 1) throw new Error('Página inválida');
      
      // LÓGICA de negócio
      const noticias = NoticiaRepository.buscarTodas(filtros);
      const total = NoticiaRepository.contar(filtros);
      
      return {
        noticias,
        total,
        paginacao: {
          pagina: filtros.pagina,
          limite: filtros.limite,
          total_páginas: Math.ceil(total / filtros.limite)
        }
      };
    }
  }

Service: Apenas Lógica ✓
- Sem req/res
- Sem banco de dados
- Testável isoladamente

  // repositories/NoticiaRepository.js
  class NoticiaRepository {
    static buscarTodas(filtros = {}) {
      let query = 'SELECT * FROM Noticia WHERE publicado = 1';
      let params = [];
      
      if (filtros.categoria) {
        query += ' AND categoriaId = ?';
        params.push(filtros.categoria);
      }
      
      const limit = filtros.limite || 10;
      const offset = ((filtros.pagina || 1) - 1) * limit;
      query += ` LIMIT ? OFFSET ?`;
      params.push(limit, offset);
      
      return db.prepare(query).all(...params);
    }
    
    static contar(filtros = {}) {
      let query = 'SELECT COUNT(*) as total FROM Noticia WHERE publicado = 1';
      let params = [];
      
      if (filtros.categoria) {
        query += ' AND categoriaId = ?';
        params.push(filtros.categoria);
      }
      
      return db.prepare(query).get(...params).total;
    }
  }

Repository: Apenas Banco ✓
- Queries parametrizadas
- Proteção contra SQL Injection
- Abstração de dados

  // utils/ApiResponse.js
  class ApiResponse {
    static success(dados, mensagem = 'Sucesso', codigo = 200) {
      return {
        status: 'success',
        code: codigo,
        data: dados,
        message: mensagem
      };
    }
    
    static error(mensagem, codigo = 400) {
      return {
        status: 'error',
        code: codigo,
        message: mensagem
      };
    }
    
    static unauthorized() {
      return {
        status: 'unauthorized',
        code: 401,
        message: 'Não autorizado'
      };
    }
  }

Factory: Respostas Padronizadas ✓
- Sempre em JSON
- Sempre com status
- Sempre com code
- Sempre com data/message
